const Rad = ({ options, value, onChange }) => (
  <div style={{ display: "flex", flexWrap: "wrap", gap: 8 }}>
    {options.map(o => (
      <button key={o} type="button" onClick={() => onChange(value === o ? "" : o)}
        style={{ padding: "8px 16px", borderRadius: 8, border: value === o ? "2px solid #2563eb" : "1px solid #d1d5db", background: value === o ? "#eff6ff" : "#fff", color: value === o ? "#1d4ed8" : "#374151", fontWeight: value === o ? 600 : 400, cursor: "pointer", fontSize: 14 }}>
        {o}
      </button>
    ))}
  </div>
);

const Chk = ({ options, value = [], onChange }) => (
  <div style={{ display: "flex", flexWrap: "wrap", gap: 8 }}>
    {options.map(o => {
      const c = value.includes(o);
      return (
        <button key={o} type="button" onClick={() => onChange(c ? value.filter(v => v !== o) : [...value, o])}
          style={{ padding: "8px 16px", borderRadius: 8, border: c ? "2px solid #2563eb" : "1px solid #d1d5db", background: c ? "#eff6ff" : "#fff", color: c ? "#1d4ed8" : "#374151", fontWeight: c ? 600 : 400, cursor: "pointer", fontSize: 14 }}>
          {c ? "✓ " : ""}{o}
        </button>
      );
    })}
  </div>
);

  const sendEmail = useCallback(() => {
    const nm = fd.name || "Employee";
    const subj = encodeURIComponent(`Migration Checklist - ${nm}`);
    const body = encodeURIComponent(genText());
    window.open(`mailto:${JP_EMAIL}?subject=${subj}&body=${body}`, "_self");
  }, [fd, genText]);

  const handleSubmit = async () => {
    const req = secs.flatMap(s => s.fields.filter(f => f.required));
    const miss = req.filter(f => { const v = fd[f.id]; return !v || (typeof v === "string" && !v.trim()); });
    if (miss.length > 0) { alert(`${t.required}\n\n${miss.map(f => `• ${f.label}`).join("\n")}`); return; }
    setStatus("submitting"); scroll();
    let sheetsOk = false;
    try {
      await fetch(SHEETS_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" }, body: JSON.stringify(genFlat()) });
      sheetsOk = true;
    } catch (e) { console.error("Sheets error:", e); }
    setTimeout(() => sendEmail(), 500);
    setStatus(sheetsOk ? "done" : "error");
  };

  const download = () => {
    const nm = fd.name || "employee";
    const b = new Blob([genText()], { type: "text/plain" });
    const u = URL.createObjectURL(b);
    const a = document.createElement("a"); a.href = u; a.download = `migration-checklist-${nm.toLowerCase().replace(/\s+/g, "-")}.txt`; a.click(); URL.revokeObjectURL(u);
  };

  const bx = { background: "#fff", borderRadius: 16, border: "1px solid #e5e7eb", padding: 24, boxShadow: "0 1px 3px rgba(0,0,0,0.04)" };
  const bt = (bg, clr, brd) => ({ padding: "12px 24px", borderRadius: 10, border: brd || "none", background: bg, color: clr, fontWeight: 600, cursor: "pointer", fontSize: 15 });

  if (status === "done" || status === "error") {
    return (
      <div style={{ maxWidth: 640, margin: "0 auto", padding: 24, fontFamily: "-apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif" }}>
        <div ref={topRef} />
        <div style={{ display: "flex", justifyContent: "flex-end", marginBottom: 16 }}>
          <button onClick={() => setLang(lang === "en" ? "ko" : "en")} style={{ ...bt("#f3f4f6","#374151","1px solid #d1d5db"), padding: "8px 16px", fontSize: 13 }}>{t.langToggle}</button>
        </div>
        {status === "done" ? (
          <div style={{ textAlign: "center", padding: "40px 24px", background: "#f0fdf4", borderRadius: 16, border: "1px solid #bbf7d0" }}>
            <div style={{ fontSize: 48, marginBottom: 12 }}>✅</div>
            <h2 style={{ fontSize: 24, fontWeight: 700, color: "#166534", margin: "0 0 8px" }}>{t.complete}</h2>
            <p style={{ color: "#4b5563", margin: "0 0 24px", lineHeight: 1.5 }}>{t.completeDesc}</p>
            <button onClick={download} style={bt("#fff","#2563eb","2px solid #2563eb")}>{t.downloadBtn}</button>
            <br /><button onClick={() => { setStatus("form"); scroll(); }} style={{ marginTop: 16, background: "none", border: "none", color: "#6b7280", cursor: "pointer", fontSize: 14 }}>{t.goBack}</button>
          </div>
        ) : (
          <div style={{ textAlign: "center", padding: "40px 24px", background: "#fef3c7", borderRadius: 16, border: "1px solid #fcd34d" }}>
            <div style={{ fontSize: 48, marginBottom: 12 }}>⚠️</div>
            <h2 style={{ fontSize: 20, fontWeight: 700, color: "#92400e", margin: "0 0 8px" }}>{t.errorTitle}</h2>
            <p style={{ color: "#78350f", margin: "0 0 24px", lineHeight: 1.5, fontSize: 14 }}>{t.errorSheets}</p>
            <div style={{ display: "flex", gap: 10, justifyContent: "center", flexWrap: "wrap" }}>
              <button onClick={download} style={bt("#fff","#2563eb","2px solid #2563eb")}>{t.downloadBtn}</button>
              <button onClick={sendEmail} style={bt("#16a34a","#fff")}>{t.emailBackup}</button>
            </div>
            <br /><button onClick={() => { setStatus("form"); scroll(); }} style={{ marginTop: 8, background: "none", border: "none", color: "#6b7280", cursor: "pointer", fontSize: 14 }}>{t.goBack}</button>
          </div>
        )}
        <div style={{ ...bx, marginTop: 24 }}>
          <h3 style={{ fontSize: 15, fontWeight: 700, color: "#dc2626", margin: "0 0 12px" }}>⚠️ {lang === "en" ? "Important Reminders Before Friday" : "금요일 전 중요 안내사항"}</h3>
          {t.reminders.map((r, i) => (<div key={i} style={{ fontSize: 13, color: "#374151", padding: "6px 0", lineHeight: 1.5 }}>• {r}</div>))}
        </div>
      </div>
    );
  }

  if (status === "submitting") {
    return (
      <div style={{ maxWidth: 640, margin: "0 auto", padding: 24, fontFamily: "-apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif" }}>
        <div ref={topRef} />
        <div style={{ textAlign: "center", padding: "60px 24px" }}>
          <div style={{ fontSize: 40, marginBottom: 16, animation: "spin 1s linear infinite" }}>⏳</div>
          <h2 style={{ fontSize: 20, fontWeight: 600, color: "#374151" }}>{t.submitting}</h2>
          <style>{`@keyframes spin { from { transform: rotate(0deg); } to { transform: rotate(360deg); } }`}</style>
        </div>
      </div>
    );
  }

  return (
    <div style={{ maxWidth: 640, margin: "0 auto", padding: "16px 16px 120px", fontFamily: "-apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif" }}>
      <div ref={topRef} />
      <div style={{ display: "flex", justifyContent: "flex-end", marginBottom: 12 }}>
        <button onClick={() => setLang(lang === "en" ? "ko" : "en")} style={{ ...bt("#f3f4f6","#374151","1px solid #d1d5db"), padding: "8px 16px", fontSize: 13 }}>{t.langToggle}</button>
      </div>
      <div style={{ textAlign: "center", marginBottom: 20 }}>
        <div style={{ fontSize: 13, fontWeight: 700, color: "#2563eb", letterSpacing: 1, textTransform: "uppercase", marginBottom: 4 }}>{t.company}</div>
        <h1 style={{ fontSize: 21, fontWeight: 700, color: "#111827", margin: "0 0 8px" }}>{t.title}</h1>
        <p style={{ fontSize: 14, color: "#6b7280", margin: 0 }}>{t.subtitle} <strong>{t.deadline}</strong></p>
      </div>
      <div style={{ background: "#fef3c7", border: "1px solid #fcd34d", borderRadius: 12, padding: 14, marginBottom: 12, fontSize: 13, color: "#92400e", lineHeight: 1.5 }}>
        <strong>⚠️</strong> {t.warning}
      </div>
      <div style={{ background: "#eff6ff", border: "1px solid #bfdbfe", borderRadius: 12, padding: 14, marginBottom: 20, fontSize: 13, color: "#1e40af", lineHeight: 1.5 }}>
        <strong>📧</strong> {t.emailBanner}
      </div>
      <div style={{ marginBottom: 16 }}>
        <div style={{ fontSize: 13, fontWeight: 600, color: "#374151", marginBottom: 6 }}>{t.sectionOf.replace("{0}", cur + 1).replace("{1}", secs.length)}</div>
        <div style={{ height: 6, background: "#e5e7eb", borderRadius: 3, overflow: "hidden" }}>
          <div style={{ height: "100%", width: `${pct}%`, background: "#2563eb", borderRadius: 3, transition: "width 0.3s" }} />
        </div>
      </div>
      <div style={{ display: "flex", gap: 4, justifyContent: "center", marginBottom: 20, flexWrap: "wrap" }}>
        {secs.map((s, i) => (
          <button key={s.id} onClick={() => go(i)} title={s.title}
            style={{ width: 32, height: 32, borderRadius: 8, border: i === cur ? "2px solid #2563eb" : "1px solid #d1d5db", background: i === cur ? "#eff6ff" : done(s) ? "#f0fdf4" : "#fff", cursor: "pointer", fontSize: 14, display: "flex", alignItems: "center", justifyContent: "center" }}>
            {s.icon}
          </button>
        ))}
      </div>
      <div style={bx}>
        <h2 style={{ fontSize: 18, fontWeight: 700, color: "#111827", margin: "0 0 20px", display: "flex", alignItems: "center", gap: 8 }}>
          <span>{sec.icon}</span> {sec.title}
        </h2>
        <div style={{ display: "flex", flexDirection: "column", gap: 20 }}>
          {sec.fields.map(f => (
            <div key={f.id}>
              <label style={{ display: "block", fontSize: 14, fontWeight: 500, color: "#374151", marginBottom: 8, lineHeight: 1.4 }}>
                {f.label}{f.required && <span style={{ color: "#dc2626", marginLeft: 4 }}>*</span>}
              </label>
              {f.type === "text" && <input type="text" value={fd[f.id] || ""} onChange={e => up(f.id, e.target.value)} placeholder={f.placeholder || ""} style={{ width: "100%", padding: "10px 14px", borderRadius: 8, border: "1px solid #d1d5db", fontSize: 14, boxSizing: "border-box", outline: "none" }} />}
              {f.type === "textarea" && <textarea value={fd[f.id] || ""} onChange={e => up(f.id, e.target.value)} placeholder={f.placeholder || ""} rows={3} style={{ width: "100%", padding: "10px 14px", borderRadius: 8, border: "1px solid #d1d5db", fontSize: 14, boxSizing: "border-box", resize: "vertical", outline: "none", fontFamily: "inherit" }} />}
              {f.type === "radio" && <Rad options={f.options} value={fd[f.id] || ""} onChange={v => up(f.id, v)} />}
              {f.type === "check" && <Chk options={f.options} value={fd[f.id] || []} onChange={v => up(f.id, v)} />}
            </div>
          ))}
        </div>
      </div>
      <div style={{ display: "flex", justifyContent: "space-between", marginTop: 20, gap: 12 }}>
        <button onClick={() => { setCur(p => p - 1); scroll(); }} disabled={cur === 0}
          style={{ ...bt(cur === 0 ? "#f3f4f6" : "#fff", cur === 0 ? "#9ca3af" : "#374151", "1px solid #d1d5db"), cursor: cur === 0 ? "default" : "pointer" }}>
          {t.back}
        </button>
        {cur < secs.length - 1 ? (
          <button onClick={() => { setCur(p => p + 1); scroll(); }} style={bt("#2563eb","#fff")}>{t.next}</button>
        ) : (
          <button onClick={handleSubmit} style={bt("#16a34a","#fff")}>{t.submit}</button>
        )}
      </div>
    </div>
  );
}
